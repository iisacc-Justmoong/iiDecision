cmake_minimum_required(VERSION 3.31)
# iisdk:consumer-application
project(iiDecisionGpuApplication LANGUAGES CXX)
include(CTest)
# This consumer project composes two separately installed SDK packages.
find_package(iiDecision 0.0.5 CONFIG REQUIRED)
find_package(iiGPU 0.1 CONFIG REQUIRED)
add_library(gpu_adapter STATIC App/GpuAdapter.cpp)
target_compile_features(gpu_adapter PUBLIC cxx_std_23)
target_include_directories(gpu_adapter PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/App")
target_link_libraries(gpu_adapter PUBLIC iiDecision::iiDecision iiGPU::iiGPU)
add_executable(iiDecisionGpuExample App/main.cpp)
target_link_libraries(iiDecisionGpuExample PRIVATE gpu_adapter)

if(BUILD_TESTING)
    set(_acceleration_tests "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/acceleration.cpp")
    set(_installed_gpu_test "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/consumer/gpu.cpp")
    if(NOT EXISTS "${_acceleration_tests}")
        set(_acceleration_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/acceleration.cpp")
        set(_installed_gpu_test "${CMAKE_CURRENT_SOURCE_DIR}/tests/gpu.cpp")
    endif()
    if(NOT EXISTS "${_acceleration_tests}")
        message(FATAL_ERROR "The GPU adapter acceptance test source is missing")
    endif()
    add_executable(iiDecisionGpuTests "${_acceleration_tests}")
    target_compile_definitions(iiDecisionGpuTests PRIVATE IIDECISION_TEST_GPU_ADAPTER)
    target_link_libraries(iiDecisionGpuTests PRIVATE gpu_adapter)
    add_executable(iiDecisionGpuConsumer "${_installed_gpu_test}")
    target_link_libraries(iiDecisionGpuConsumer PRIVATE gpu_adapter)
    foreach(backend metal vulkan)
        add_test(NAME "iiDecision.Acceleration.${backend}" COMMAND iiDecisionGpuTests Devices "${backend}/")
        set_tests_properties("iiDecision.Acceleration.${backend}" PROPERTIES TIMEOUT 120 SKIP_RETURN_CODE 77)
        add_test(NAME "iiDecision.Acceleration.${backend}_reuse" COMMAND iiDecisionGpuTests Reuse "${backend}/")
        set_tests_properties("iiDecision.Acceleration.${backend}_reuse" PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77)
        add_test(NAME "iiDecision.InstalledGpu.${backend}" COMMAND iiDecisionGpuConsumer "${backend}/")
        set_tests_properties("iiDecision.InstalledGpu.${backend}" PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77)
    endforeach()
    add_test(NAME iiDecision.Acceleration.shared_gpu_concurrency COMMAND iiDecisionGpuTests SharedGpuConcurrency)
    set_tests_properties(iiDecision.Acceleration.shared_gpu_concurrency PROPERTIES TIMEOUT 120 SKIP_RETURN_CODE 77)
endif()
